# Semantic-repetition metrics only.  Kept out of requirements.txt because
# sentence-transformers pulls in torch, which is a multi-gigabyte install that
# most TextGrader users do not need.
#
# Without it, the four `semantic_*` metrics fall back to a TF-IDF lexical
# overlap.  That is a different quantity from an embedding similarity, and both
# the finding's warning and its evidence say which backend produced the number.
-r requirements.txt
sentence-transformers>=2.2

# NLI cross-encoder, real semantic-role-labelling, closed-schema relation
# extraction and real argument-relation classification (logic_suite's
# `nli_entailment`, `semantic_role_labeling`, `relation_extraction` and
# `argument_mining`), coreference resolution (logic_suite and coherence_suite),
# and a causal language model (randomness_suite's `neural_language_model`).
# All of these share the torch install above and every one is off by default;
# without them each channel reports "unavailable" with an install hint and its
# suite's other measurements are unaffected. None of the four logic_suite
# model channels needs a package beyond `transformers` itself -- each is just
# a different Hugging Face hub checkpoint, fetched on first use the same way
# `nli_entailment`'s already is.
#
# fastcoref as shipped does not do the tied-weight bookkeeping transformers 5.x
# expects; textgrader.optional.shim_fastcoref_transformers supplies the missing
# default, and both the coherence and logic suites load coreference through it,
# so the two cannot disagree about whether it is available.  If that stops being
# enough, coreference degrades to "unavailable" rather than crashing.
transformers>=4.40
torch>=2.0
fastcoref>=2.1

# Real RST (Rhetorical Structure Theory) discourse-tree parsing
# (coherence_suite's `rst` feature, off by default). Shares the torch/
# transformers install above; also needs its own non-PyPI `isanlp`
# dependency, not installable from this file since it is not on PyPI:
#     pip install git+https://github.com/iinemo/isanlp.git
# Constructing the parser downloads ~5 GB of model checkpoints on first use
# and holds ~6 GB of RAM resident while loaded -- see coherence_suite's
# module docstring and config.json's `_requires_rst` note for the full cost
# story and the sampling this feature uses to bound it. Without isanlp_rst
# (or isanlp), every `discourse.coherence_rst_*` metric id reports
# "unavailable" with an install hint and the rest of the suite is
# unaffected.
isanlp-rst>=3.2

# Real constituency-tree parsing (syntax_complexity_suite's `constituency`
# feature, off by default even when the suite is on). Shares the torch
# install above; the pretrained checkpoint itself (~260 MB, `benepar_en3`)
# is a separate one-time download, not part of the pip package:
#     python -c "import benepar; benepar.download('benepar_en3')"
# Without it, every `syntax.complexity_constituency_*` metric id reports
# "unavailable" with an install hint; every other feature in the suite is
# spaCy-only and unaffected. A real transformers/benepar version-skew bug
# (T5Tokenizer missing build_inputs_with_special_tokens under
# transformers>=5) is worked around in-module; see that module's docstring.
benepar>=0.2
